////////////////////////////////////////////////////////////
// CORE I18N AND IDENTITY
////////////////////////////////////////////////////////////

Table languages {
  id uuid [pk]
  iso_code varchar(10) [not null, unique, note: 'ISO 639-1 or BCP-47, e.g. en, fr, am']
  english_name varchar(120) [not null]
  native_name varchar(120)
  is_active boolean [not null, default: true]
  created_at timestamptz [not null, default: `now()`]
}

Table users {
  id uuid [pk]
  username varchar(60) [not null, unique]
  email varchar(255) [not null, unique]
  role app_role [not null, default: 'student']
  ui_language_id uuid [not null, ref: > languages.id]
  base_language_id uuid [not null, ref: > languages.id, note: 'Language user learns with']
  target_language_id uuid [not null, ref: > languages.id, note: 'Usually French for your app']
  current_level proficiency_level [not null, default: 'a0']
  xp_total int [not null, default: 0]
  streak_days int [not null, default: 0]
  is_active boolean [not null, default: true]
  created_at timestamptz [not null, default: `now()`]
  updated_at timestamptz [not null, default: `now()`]
}

Table user_profiles {
  user_id uuid [pk, ref: > users.id]
  display_name varchar(120)
  avatar_url text
  country_code varchar(2)
  timezone varchar(60) [not null, default: 'UTC']
  birth_year int
  bio text
  preferred_voice_id uuid [ref: > tts_voices.id]
  learning_goal text
  created_at timestamptz [not null, default: `now()`]
  updated_at timestamptz [not null, default: `now()`]
}

Table subscriptions {
  id uuid [pk]
  user_id uuid [not null, ref: > users.id]
  tier subscription_tier [not null, default: 'free']
  provider varchar(50)
  provider_customer_id varchar(120)
  provider_subscription_id varchar(120)
  status varchar(40) [not null, default: 'active']
  starts_at timestamptz [not null, default: `now()`]
  ends_at timestamptz
  created_at timestamptz [not null, default: `now()`]

  indexes {
    (user_id, status)
  }
}
